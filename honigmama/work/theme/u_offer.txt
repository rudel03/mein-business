{%- comment -%}
  Paket-Angebot (Festpreis-Pakete, kein Rabatt – Buchpreisbindung).
  mode: 'cart'    -> im Warenkorb, tauscht Einzelbände gegen das Paket
  mode: 'product' -> unter dem Kauf-Button, Parameter current: product.handle
{%- endcomment -%}
{%- liquid
  assign h1 = 'honigmama-band-1'
  assign h2 = 'honigmama-band-2'
  assign h3 = 'honigmama-band-3'
  assign hp = 'honigmama-paket-band-1-2'
  assign hk = 'honigmama-komplett'
  assign p1 = all_products[h1]
  assign p2 = all_products[h2]
  assign p3 = all_products[h3]
  assign pp = all_products[hp]
  assign pk = all_products[hk]
  assign v1 = p1.selected_or_first_available_variant
  assign v2 = p2.selected_or_first_available_variant
  assign v3 = p3.selected_or_first_available_variant
  assign vp = pp.selected_or_first_available_variant
  assign vk = pk.selected_or_first_available_variant

  assign has1 = false
  assign has2 = false
  assign has3 = false
  assign hasp = false
  assign hask = false
  assign mode = mode | default: 'product'
  if mode == 'cart'
    for line in cart.items
      case line.product.handle
        when h1
          assign has1 = true
        when h2
          assign has2 = true
        when h3
          assign has3 = true
        when hp
          assign hasp = true
          assign has1 = true
          assign has2 = true
        when hk
          assign hask = true
      endcase
    endfor
  else
    case current
      when h1
        assign has1 = true
      when h2
        assign has2 = true
      when h3
        assign has3 = true
      when hp
        assign hasp = true
        assign has1 = true
        assign has2 = true
      when hk
        assign hask = true
    endcase
  endif

  assign offer = ''
  unless hask
    if has1 and has2 and has3
      assign offer = 'k'
    elsif has1 and has2
      assign offer = 'k'
    elsif has3
      assign offer = 'k'
    elsif has1 or has2
      assign offer = 'p'
    endif
  endunless
  if offer == 'p' and pp.available != true
    assign offer = ''
  endif
  if offer == 'k' and pk.available != true
    assign offer = ''
  endif

  # Einzelpreis-Summe der Bände im Paket und schon vorhandener Anteil
  if offer == 'p'
    assign pack = pp
    assign pv = vp
    assign single_sum = v1.price | plus: v2.price
    assign pack_name = 'Band 1 + 2'
    assign missing = ''
    unless has1
      assign missing = 'Band 1 · Was darf ich essen?'
    endunless
    unless has2
      assign missing = 'Band 2 · Vorkochen fürs Wochenbett'
    endunless
  elsif offer == 'k'
    assign pack = pk
    assign pv = vk
    assign single_sum = v1.price | plus: v2.price | plus: v3.price
    assign pack_name = 'Band 1 + 2 + 3'
    assign missing = ''
    unless has1
      assign missing = missing | append: 'Band 1, '
    endunless
    unless has2
      assign missing = missing | append: 'Band 2, '
    endunless
    unless has3
      assign missing = missing | append: 'Band 3, '
    endunless
    assign missing = missing | remove_last: ', '
  endif

  assign owned_sum = 0
  if has1 and hasp == false
    assign owned_sum = owned_sum | plus: v1.price
  endif
  if has2 and hasp == false
    assign owned_sum = owned_sum | plus: v2.price
  endif
  if has3
    assign owned_sum = owned_sum | plus: v3.price
  endif
  if hasp
    assign owned_sum = owned_sum | plus: vp.price
  endif
  assign extra = pv.price | minus: owned_sum
  assign saving = single_sum | minus: pv.price
-%}
