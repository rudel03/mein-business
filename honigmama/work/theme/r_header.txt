<script>
(function(){
  var d=document.getElementById('hm-drawer');if(!d)return;
  var openers=document.querySelectorAll('[data-hm-drawer-open]'),last=null;
  function set(v){openers.forEach(function(b){b.setAttribute('aria-expanded',v)})}
  function open(e){last=e&&e.currentTarget;d.hidden=false;requestAnimationFrame(function(){d.classList.add('is-open')});set(true);document.documentElement.style.overflow='hidden';var x=d.querySelector('.hm-drawer__x');x&&x.focus()}
  function close(){d.classList.remove('is-open');set(false);document.documentElement.style.overflow='';setTimeout(function(){d.hidden=true},350);last&&last.focus()}
  openers.forEach(function(b){b.addEventListener('click',open)});
  d.querySelectorAll('[data-hm-drawer-close]').forEach(function(b){b.addEventListener('click',close)});
  d.querySelectorAll('a[href*="#"]').forEach(function(a){a.addEventListener('click',close)});
  document.addEventListener('keydown',function(e){if(e.key==='Escape'&&!d.hidden)close()});
})();
</script>
